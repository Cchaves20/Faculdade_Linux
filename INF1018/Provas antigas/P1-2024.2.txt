.globl corta
corta:
	pushq %rbp
	movq %rsp, %rbp
	addl %edi, %esi
	movl %esi, %eax
	leave
	ret

.globl buba
buba:
	pushq %rbp
	movq %rsp, %rbp
	movl $0, %eax
	
for:
	cmpl %eax, %esi
	jg fim
	
	cmpl %rdi(eax), %dl
	jge if
	
	jmp skip

if:
	
	addl %eax call corta
	
skip:
